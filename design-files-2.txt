import useInView from '../hooks/useInView.js'

/**
 * Wraps children in a fade/rise-on-scroll animation.
 * Usage: <Reveal delay={100}><h2>Title</h2></Reveal>
 */
export default function Reveal({ children, as: Tag = 'div', delay = 0, className = '', ...rest }){
  const [ref, inView] = useInView()

  return (
    <Tag
      ref={ref}
      className={`reveal ${inView ? 'is-visible' : ''} ${className}`}
      style={{ transitionDelay: inView ? `${delay}ms` : '0ms' }}
      {...rest}
    >
      {children}
    </Tag>
  )
}

import { useState } from 'react';
import { leadershipTerms } from '../data/leadership';
import './Leadership.css';

function getInitials(name) {
  return name
    .split(' ')
    .filter(Boolean)
    .slice(0, 2)
    .map((part) => part[0].toUpperCase())
    .join('');
}

function MemberCard({ member, termId }) {
  const src = `/images/leadership/${termId}/${member.image}`;
  const [failedSrc, setFailedSrc] = useState(null);
  const imageFailed = failedSrc === src;

  return (
    <article className="leader-card">
      <div className="leader-photo">
        {imageFailed ? (
          <span className="leader-initials">{getInitials(member.name)}</span>
        ) : (
          <img
            src={src}
            alt={member.name}
            loading="lazy"
            onError={() => setFailedSrc(src)}
          />
        )}
      </div>
      <h3 className="leader-name">{member.name}</h3>
      <p className="leader-role">{member.role}</p>
      {member.note && <p className="leader-note">{member.note}</p>}
    </article>
  );
}

export default function Leadership() {
  const [activeId, setActiveId] = useState(leadershipTerms[0].id);
  const activeTerm = leadershipTerms.find((t) => t.id === activeId);

  return (
    <div className="leadership-page">
      <header className="leadership-hero">
        <h1>Leadership</h1>
        <p>The office bearers who have guided TMSK through each term.</p>
      </header>

      <nav className="term-tabs" aria-label="Leadership terms">
        {leadershipTerms.map((term) => (
          <button
            key={term.id}
            type="button"
            className={term.id === activeId ? 'term-tab active' : 'term-tab'}
            onClick={() => setActiveId(term.id)}
          >
            {term.title}
          </button>
        ))}
      </nav>

      <section className="term-section" aria-labelledby={`${activeTerm.id}-title`}>
        <h2 id={`${activeTerm.id}-title`} className="term-title">
          {activeTerm.title}
        </h2>
        <div className="leader-grid">
          {activeTerm.members.map((member) => (
            <MemberCard
              key={`${activeTerm.id}-${member.role}`}
              member={member}
              termId={activeTerm.id}
            />
          ))}
        </div>
      </section>
    </div>
  );
}

import { useEffect, useState } from 'react'
import './EventPopup.css'

const SEEN_KEY = 'tmsk_event_popup_seen'
const REGISTER_URL = 'https://docs.google.com/forms/d/1mbJFR53KfgaMFuwOZ7NbxB3_t9-_45dYc51Xy_QvIE8/edit'

export default function EventPopup(){
  const [visible, setVisible] = useState(false)

  useEffect(() => {
    if (sessionStorage.getItem(SEEN_KEY)) return
    const timer = setTimeout(() => setVisible(true), 900)
    return () => clearTimeout(timer)
  }, [])

  const dismiss = () => {
    setVisible(false)
    sessionStorage.setItem(SEEN_KEY, '1')
  }

  if (!visible) return null

  return (
    <div className="event-popup-overlay" onClick={dismiss}>
      <div className="event-popup" role="dialog" aria-label="Event registration" onClick={e => e.stopPropagation()}>
        <button className="event-popup-close" aria-label="Close" onClick={dismiss}>&times;</button>
        <span className="event-popup-tag">Registration open</span>
        <p className="event-popup-title">TMSK Art Online Competition</p>
        <p className="event-popup-sub">Registration closes 31 Aug 2026 &middot; Entry fee ₹75</p>
        <a
          href={REGISTER_URL}
          target="_blank"
          rel="noopener noreferrer"
          className="btn event-popup-btn"
          onClick={dismiss}
        >
          Register Now
        </a>
      </div>
    </div>
  )
}
